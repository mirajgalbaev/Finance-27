import 'dart:io';

void main() {
  var file = File('lib/ui/screens/planner_screen.dart');
  var lines = file.readAsLinesSync();
  // We want to delete 1597-1606 and 1615-2319
  var newLines = <String>[];
  for (int i = 0; i < lines.length; i++) {
     int lineNum = i + 1;
     if (lineNum >= 1597 && lineNum <= 1606) continue;
     if (lineNum >= 1615 && lineNum <= 2319) continue;
     newLines.add(lines[i]);
  }
  file.writeAsStringSync(newLines.join('\n') + '\n');
  print('Done purging lines');
}
